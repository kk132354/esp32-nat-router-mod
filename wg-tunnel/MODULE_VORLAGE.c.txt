/*
 * VORLAGE fuer ein neues Modul.
 *
 * So geht's:
 *   1. Diese Datei kopieren nach  main/mod_<name>.c   (in BEIDEN Projekten, falls noetig)
 *   2. Name und Funktionen anpassen
 *   3. einmal "idf.py reconfigure", dann "idf.py build"
 * main.c muss nicht angefasst werden.
 *
 * Einstellungen: Entweder in main/Kconfig.projbuild ein neues "config MEINMODUL_..."
 * ergaenzen, oder (einfacher) hier mit #ifndef einen Standardwert festlegen.
 */
#include <stdio.h>

#include "esp_http_server.h"
#include "esp_log.h"
#include "module_api.h"
#include "web.h"

#ifndef CONFIG_MEINMODUL_INTERVAL_S
#define CONFIG_MEINMODUL_INTERVAL_S 10
#endif

static const char *TAG = "meinmodul";
static int s_zaehler = 0;

static void meinmodul_init(void)
{
    ESP_LOGI(TAG, "Modul laeuft, Intervall %d s", CONFIG_MEINMODUL_INTERVAL_S);
    /* hier z. B. GPIO einrichten oder einen eigenen Task starten: xTaskCreate(...) */
}

/* optional: Zeilen fuer die Statusseite des DevKits */
static int meinmodul_status(char *buf, size_t len)
{
    return snprintf(buf, len, "Mein Modul:      Zaehler %d\n", s_zaehler);
}

/* optional: eigene Webseite  http://<IP>/meinmodul  (erscheint automatisch im Menue) */
static esp_err_t meinmodul_page(httpd_req_t *req)
{
    char body[200];
    snprintf(body, sizeof(body), "<h2>Mein Modul</h2><p>Zaehler: %d</p>%s",
             s_zaehler, web_can_control(req) ? "<p>Du darfst aendern.</p>" : "<p>Nur ansehen (bitte anmelden).</p>");
    return web_send_page(req, "Mein Modul", body);
}

static void meinmodul_http(httpd_handle_t srv)
{
    httpd_uri_t u = {.uri = "/meinmodul", .method = HTTP_GET, .handler = meinmodul_page};
    httpd_register_uri_handler(srv, &u);
}

/* optional: Pinbelegung. Erscheint auf der Seite "Pins" und auf der Seite des Moduls.
 * gpio = -1 ist nur ein Hinweis (Strom, Bauteil). Doppelt belegte Pins werden rot gemeldet. */
static size_t meinmodul_pins(module_pin_t *out, size_t max)
{
    static const module_pin_t list[] = {
        {5, true, "Signal fuer mein Bauteil (Ausgang)"},
        {-1, false, "Bauteil Plus an 3,3 V, Minus an GND"},
    };
    size_t n = sizeof(list) / sizeof(list[0]);
    if (n > max) {
        n = max;
    }
    for (size_t i = 0; i < n; i++) {
        out[i] = list[i];
    }
    return n;
}

/* Stufe: MODULE_STAGE_EARLY (vor WLAN), MODULE_STAGE_NETWORK (WLAN+Uhrzeit da),
 *        MODULE_STAGE_TUNNEL (WireGuard eingerichtet). priority: kleiner = frueher. */
MODULE_REGISTER(meinmodul,
                .name = "meinmodul",
                .stage = MODULE_STAGE_NETWORK,
                .priority = 50,
                .init = meinmodul_init,
                .status = meinmodul_status,
                .http = meinmodul_http,
                .menu_title = "Mein Modul",
                .menu_path = "meinmodul",
                .pins = meinmodul_pins)
